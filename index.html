Jovas Language
Reference

Jovas is an AI-native backend programming language designed to combine the best features of all major languages into one clean, expressive syntax — with AI, databases, deployment, security, and real-time communication built in from day one.

Version

This documentation covers Jovas v1.0.0. The language is actively developed.

Installation

Jovas requires Python 3.10 or higher. No external packages are needed for the core language.

# Clone the repository
git clone https://github.com/jovas-lang/jovas
cd jovas

# Run the demo to verify installation
python run.py

# Start the interactive REPL
python run.py repl

Quick Start

Create a file called app.jovas and run it:

// app.jovas
const APP  = "MyApp"
const PORT = 8080

fn onStart()
    log.info(["${APP} running on port ${PORT}"])

let app = server.create()
app.listen(PORT, onStart)
python run.py run app.jovas

Variables & Constants

Use let for mutable variables and const for immutable constants. Jovas is dynamically typed — no type annotations required (though optional type hints are supported).

let name    = "Jovas"      // string
let port    = 8080         // integer
let active  = true         // boolean
let data    = null         // null
const MAX   = 9999         // immutable — cannot be reassigned

// String interpolation with ${}
let msg = "Hello, ${name}! Port: ${port}"

// Null-safe access with ?.
let avatar = user?.profile?.avatar

Functions

// Basic function
fn greet(name)
    return "Hello, ${name}!"

// Async function
async fn fetchData(url)
    let res = await http.get(url)
    return res.body

// Expose as REST endpoint automatically
expose fn getUser(id)
    return db.findOne("users", { id: id })

// Optional type hints
fn add(a: int, b: int): int
    return a + b

Multiple return values

fn minmax(items)
    return min(items), max(items)

let lo, hi = minmax([3, 1, 9, 2])

Classes

network Hacker Toolkit

Port scanning, DNS lookup, WHOIS, ping, and payload generation for network analysis and ethical pen testing.

// Port scan
let ports = network.scan("192.168.1.1", [80, 443, 22, 3306])
print(ports[80])      // "open" or "closed"

// Ping
let r = network.ping("8.8.8.8")
print(r.alive)        // true
print(r.latency)      // 12 (ms)

// DNS lookup
let dns = network.dns("example.com")
print(dns.ip)         // "93.184.216.34"

// WHOIS
let w = network.whois("example.com")
print(w.registrar)

// HTTP request
let res = network.get("https://api.example.com/data")
print(res.status)     // 200

// Response headers
let h = network.headers("https://example.com")
print(h.server)       // "nginx"

// Payload generation (pen testing)
let sqli = network.payload("sqli")   // SQL injection payloads
let xss  = network.payload("xss")    // XSS payloads
let lfi  = network.payload("path")   // Path traversal
network.scan(host, ports[]) — Scan ports, returns {port: "open"|"closed"}
network.ping(host) — Ping host, returns {host, alive, latency}
network.dns(host) — DNS lookup, returns {domain, ip, mx, ns}
network.whois(domain) — WHOIS data
network.get(url) — HTTP GET request
network.headers(url) — Fetch response headers
network.payload(type) — Generate test payloads: sqli, xss, path, cmd

game 2D Engine

Built-in 2D game engine. Create sprites, apply physics, detect collisions and run a game loop — all in Jovas.

// Setup
let canvas = game.canvas(800, 600)
let hero   = game.sprite("hero",  100, 300, 32, 32)
let enemy  = game.sprite("enemy", 600, 300, 32, 32)

// Game loop
fn onTick()
    game.clear("#000")
    game.move(hero, 2, 0)           // move right
    game.physics(hero, 0.5)         // apply gravity
    if game.collide(hero, enemy)
        game.score(100)
    game.text("Score: 100", 10, 20)
    game.render()

game.loop(onTick)

// Export to HTML5
let out = game.export()
print(out.url)   // "game.html"
game.canvas(w, h) — Create game canvas
game.sprite(name, x, y, w, h) — Create sprite
game.clear(color) — Clear canvas
game.move(sprite, dx, dy) — Move sprite
game.physics(sprite, gravity) — Apply gravity
game.collide(s1, s2) — Collision detection
game.text(text, x, y) — Draw text
game.render() — Draw all sprites
game.loop(fn) — Start game loop
game.score(n) — Update score
game.export() — Export to HTML5

story Novel Writing

Write novels, scripts and stories in any language. Structure chapters, define characters, generate AI prose.

// Create a novel
let novel = story.create("The Last Code", "sci-fi thriller")
let alice = story.character("Alice", "hacker", ["brave", "resourceful"])
let bob   = story.character("Bob", "villain", ["cunning"])

// Add chapters and scenes
let ch1 = story.chapter(novel, "Chapter 1", "The breach")
let sc1 = story.scene(ch1, "Alice discovers the server room at midnight")

// Dialogue
story.dialogue(alice, "They got in. All of it.")
story.dialogue(bob,   "You are already too late.")

// AI-powered writing
let prose = ai.write("Tense confrontation in a dark server room — sci-fi thriller")
print(prose)

// Translate entire novel
let french = story.translate(novel, "French")

// Stats and export
print(story.wordCount(novel))
let epub = story.export(novel, "epub")
print(epub.url)    // "the-last-code.epub"
story.create(title, genre) — Create novel object
story.chapter(novel, title, text) — Add chapter
story.scene(chapter, description) — Add scene
story.character(name, role, traits[]) — Define character
story.dialogue(character, text) — Add dialogue
story.translate(novel, lang) — Translate to any language
story.wordCount(novel) — Count words
story.export(novel, format) — Export as text/pdf/epub
story.outline(idea, options) — Generate story outline

biometric Auth

Fingerprint, face, and voice authentication. Supports MFA (multi-factor authentication) combining multiple biometric types.

// Single biometric
let bio = biometric.init({type: "fingerprint"})
bio.enroll("user-alice")
let r = bio.verify("user-alice", {type: "fingerprint"})
print(r.authenticated)   // true

// Multi-factor authentication
let mfa = biometric.mfa(["fingerprint", "face"])
mfa.enroll("user-alice")
let result = mfa.verify("user-alice")
print(result.authenticated)   // true
print(result.methods)         // ["fingerprint", "face"]

// Check enrollment
let status = biometric.isEnrolled("user-alice")
print(status)   // true
biometric.init({type}) — Init biometric (fingerprint/face/voice)
biometric.enroll(userId) — Enroll a user
biometric.verify(userId, options) — Verify identity
biometric.mfa(types[]) — Create MFA with multiple types
biometric.isEnrolled(userId) — Check enrollment status

POST /api/completions — Server-side autocomplete
POST /api/share — Create shareable code snippet
GET /api/health — Health check + interpreter status
GET /api/examples — All 18 examples metadata
